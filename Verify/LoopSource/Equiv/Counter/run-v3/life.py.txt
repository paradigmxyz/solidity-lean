import sys,json,subprocess,importlib.util,hashlib
sys.path.insert(0,'/home/hand/solar446/tools/certify')
spec=importlib.util.spec_from_file_location('g','/home/hand/solar446/tools/certify/generic.py');g=importlib.util.module_from_spec(spec);spec.loader.exec_module(g)
import keccak
S='/home/hand/exporter/target/debug/solar'
o=subprocess.run([S,'/home/hand/run/Counter.sol','-Ogas','--evm-version','osaka','--emit=bin,bin-runtime,abi'],capture_output=True,text=True)
j=json.loads(o.stdout); k=[k for k in j['contracts'] if k.endswith(':Counter')][0]; c=j['contracts'][k]
init=bytes.fromhex(c['bin']); plain=bytes.fromhex(c['bin-runtime'])
ws=[int(a) for a in sys.argv[1].split(',')]; storage=json.loads(sys.argv[2]); out=sys.argv[3]; ns=sys.argv[4]
abi=c['abi'] if isinstance(c['abi'],list) else json.loads(c['abi'])
